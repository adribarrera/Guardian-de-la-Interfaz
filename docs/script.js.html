// Elementos del formulario y la interfaz
const contenedorBienvenida = document.getElementById("contenedor-bienvenida");
const inputNombre = document.getElementById("nombreUsuario");
const inputColor = document.getElementById("colorFondo");
const selectTema = document.getElementById("seleccionarTema");
const btnGuardar = document.getElementById("btnGuardar");
const btnReset = document.getElementById("btnReset");

// Cambio de color de fondo
inputColor.addEventListener("input", () => {
    document.body.style.backgroundColor = inputColor.value;
});

// Cambio de tema
selectTema.addEventListener("change", () => {
    if (selectTema.value === "oscuro") {
        document.body.classList.add("tema-oscuro");
    } else {
        document.body.classList.remove("tema-oscuro");
    }
});

/**
 * Guarda una cookie con nombre, valor y dias de expiracion (opcionales)
 * @param {string} nombre - Nombre de la cookie
 * @param {string} valor - Valor a almacenar
 * @param {number|null} dias - Dias de duracion. Si es null, es de sesion
 */
function setCookie(nombre, valor, dias) {
    let expiracion = "";
    if (dias) {
        const fecha = new Date();
        fecha.setTime(fecha.getTime() + (dias * 24 * 60 * 60 * 1000));
        expiracion = "; expires=" + fecha.toUTCString();
    }
    document.cookie = encodeURIComponent(nombre) + "=" + encodeURIComponent(valor) + expiracion + "; path=/";
}

/**
 * Lee el valor de una cookie guardada por su nombre
 * @param {string} nombre - Nombre de la cookie a buscar
 * @returns {string|null} Valor decodificado o null si no existe
 */
function getCookie(nombre) {
    const pares = document.cookie.split("; ");

    for (let par of pares) {
        const [clave, valor] = par.split("=");
        if (clave === encodeURIComponent(nombre)) {
            return decodeURIComponent(valor);
        }
    }
    return null;
}

/**
 * Elimina una cookie asignandole una fecha del pasado
 * @param {string} nombre - Nombre de la cookie a eliminar
 */
function deleteCookie(nombre) {
    document.cookie = encodeURIComponent(nombre) + "=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/";
}

/**
 * Genera e inserta el bloque de bienvenida en el DOM con createElement
 * @param {string} nombre - Nombre del usuario a mostrar
 */
function mostrarBienvenida(nombre) {
    // Vacia el contenedor
    while (contenedorBienvenida.firstChild) {
        contenedorBienvenida.removeChild(contenedorBienvenida.firstChild);
    }

    const titulo = document.createElement("h2");
    titulo.textContent = "¡Hola de nuevo, " + nombre + "!";

    const parrafo = document.createElement("p");
    parrafo.textContent = "Tus preferencias de interfaz se han cargado automáticamente.";

    contenedorBienvenida.appendChild(titulo);
    contenedorBienvenida.appendChild(parrafo);
}

// Guarda preferencias
btnGuardar.addEventListener("click", () => {
    const nombre = inputNombre.value.trim();
    const color = inputColor.value;
    const tema = selectTema.value;

    if (!nombre) {
        alert("Por favor, escribe tu nombre antes de guardar.");
        inputNombre.focus();
        return;
    }

    // Guarda cookies: 7 dias para nombre y color, de sesion para tema
    setCookie("nombreUsuario", nombre, 7);
    setCookie("colorFondo", color, 7);
    setCookie("tema", tema, null);

    // Actualiza mensaje en pantalla
    mostrarBienvenida(nombre);

    alert("Preferencias guardadas correctamente.");
});

// Restaura configuracion al cargar la pagina
window.addEventListener("DOMContentLoaded", () => {
    const usuarioGuardado = getCookie("nombreUsuario");
    const colorGuardado = getCookie("colorFondo");
    const temaGuardado = getCookie("tema");

    if (usuarioGuardado) {
        inputNombre.value = usuarioGuardado;
        mostrarBienvenida(usuarioGuardado);
    }

    if (colorGuardado) {
        inputColor.value = colorGuardado;
        document.body.style.backgroundColor = colorGuardado;
    }

    if (temaGuardado) {
        selectTema.value = temaGuardado;
        if (temaGuardado === "oscuro") {
            document.body.classList.add("tema-oscuro");
        } else {
            document.body.classList.remove("tema-oscuro");
        }
    }
});

// Restablece interfaz al hacer click
btnReset.addEventListener("click", () => {
    // Borra cookies
    deleteCookie("nombreUsuario");
    deleteCookie("colorFondo");
    deleteCookie("tema");

    // Elimina mensaje de bienvenida
    while (contenedorBienvenida.firstChild) {
        contenedorBienvenida.removeChild(contenedorBienvenida.firstChild);
    }

    // Restaura valores por defecto de los campos
    inputNombre.value = "";
    inputColor.value = "#F4EFE6";
    selectTema.value = "claro";

    // Quita estilos en linea y clases
    document.body.style.backgroundColor = "";
    document.body.classList.remove("tema-oscuro");

    alert("Preferencias reiniciadas con éxito.");
});
